llowed];
    DB.forEach(f=>{
      if(!expanded.some(x=>x.id===f.id)){
        const useful=(f.fi>4||f.p>15||f.cat==='carb'||f.cat==='fat');
        if(useful)expanded.push(f);
      }
    });
    res=optimize(expanded,state.targets);
  }
  state.result=res;
  renderMenu(unknown);
  goStep(3);
}

function renderMenu(unknown=[]){
  const res=state.result, t=state.targets;
  qs('menuIntro').textContent=state.mode==='strict'
    ? 'נבנה רק מהמאכלים שכתבת. שום מאכל חדש לא נכנס ללא אישור.'
    : 'המאכלים שכתבת קיבלו עדיפות, והמערכת השלימה לפי הצורך.';

  const chosen=res.foods.map((f,i)=>({f,q:res.amounts[i]})).filter(x=>x.q>0);
  const meals=[{name:'ארוחת בוקר',items:[]},{name:'ארוחת צהריים',items:[]},{name:'ארוחת ערב',items:[]},{name:'נשנוש',items:[]}];
  chosen.forEach((x,i)=>{
    let idx=i%4;
    if(x.f.cat==='protein') idx=[0,1,2][i%3];
    if(x.f.cat==='fat') idx=(i%2===0)?0:2;
    if(x.f.cat==='fiber'&&i%3===0) idx=3;
    meals[idx].items.push(x);
  });

  const wrap=qs('meals');wrap.innerHTML='';
  meals.forEach((m,mi)=>{
    if(!m.items.length)return;
    const card=document.createElement('div');card.className='meal';
    card.innerHTML=`<div class="mealhead"><h3>${m.name}</h3><button class="mini" onclick="shuffleMeal(${mi})">הצע ארוחה אחרת</button></div>`;
    m.items.forEach(({f,q})=>{
      const row=document.createElement('div');row.className='foodrow';
      row.innerHTML=`<div><strong>${f.name}</strong><div class="muted">${humanQuantity(f,q)}</div>${f.source?`<span class="source-tag">${f.source}</span>`:''}</div><button class="mini">ערכים</button><button class="mini">החלף</button><button class="mini">כמות</button>`;
      const btns=row.querySelectorAll('button');
      btns[0].onclick=()=>showFoodValues(f);
      btns[1].onclick=()=>replaceFood(f.id);
      btns[2].onclick=()=>changeQty(f.id);
      card.appendChild(row);
    });
    wrap.appendChild(card);
  });

  const issues=issueList(res);
  const imb=qs('imbalance');imb.innerHTML='';
  if(issues.length){
    const box=document.createElement('div');box.className='note warn';
    box.innerHTML=`<strong>זיהיתי חוסר איזון:</strong><br>${issues.join(' · ')}`;
    imb.appendChild(box);
    if(state.mode==='strict'){
      const ids=enteredFoods().map(f=>f.id);
      const sug=bestFiberSuggestions(ids);
      const add=document.createElement('div');add.className='note info';
      add.innerHTML=`<strong>אפשר לתקן בלי לכפות מאכל חדש.</strong><br>לדוגמה, אם הבעיה היא סיבים מול קלוריות, אפשר לשקול: ${sug.map(f=>f.name).join(' / ')}.<br><button class="mini" style="margin-top:8px">הצע לי תיקון ממוקד</button>`;
      add.querySelector('button').onclick=()=>showFocusedFix(sug);
      imb.appendChild(add);
    }
  }else{
    imb.innerHTML='<div class="note success"><strong>איזון טוב:</strong> התפריט קרוב ליעדים שהוגדרו.</div>';
  }
  if(unknown.length){
    imb.innerHTML+=`<div class="note warn">לא הצלחתי לזהות: ${unknown.map(x=>x.raw).join(', ')}. הם לא נכנסו לחישוב.</div>`;
  }

  renderBalance();
}

function renderBalance(){
  const a=state.result.totals,t=state.targets;
  const rows=[
    ['קלוריות','cal',t.cal,''],
    ['חלבון','protein',t.protein,' ג׳'],
    ['פחמימות','carbs',t.carbs,' ג׳'],
    ['שומן','fat',t.fat,' ג׳'],
    ['סיבים','fiber',t.fiber,' ג׳']
  ];
  qs('balance').innerHTML=rows.map(([label,k,goal,u])=>{
    const val=Math.round(a[k]);const pct=Math.min(130,Math.round(val/goal*100));
    return `<div class="barbox"><strong>${label}</strong><div>${val}${u} / ${Math.round(goal)}${u}</div><div class="barline"><div class="barfill" style="width:${Math.min(100,pct)}%"></div></div><small>${pct}%</small></div>`;
  }).join('');
}

function replaceFood(id){
  const res=state.result;
  const idx=res.foods.findIndex(f=>f.id===id); if(idx<0)return;
  const old=res.foods[idx];
  const allowed=state.mode==='strict'?enteredFoods():DB;
  const candidates=allowed.filter(f=>f.id!==id&&(f.cat===old.cat||Math.abs(f.p-old.p)<8));
  if(!candidates.length){alert('אין כרגע חלופה מותרת למאכל הזה.');return;}
  const next=candidates[Math.floor(Math.random()*candidates.length)];
  const newFoods=[...res.foods];newFoods[idx]=next;
  state.result=optimize(newFoods,state.targets);
  renderMenu();
}
function changeQty(id){
  const res=state.result,idx=res.foods.findIndex(f=>f.id===id);if(idx<0)return;
  const current=Math.round(res.amounts[idx]);
  const val=prompt(`כמות חדשה בגרמים עבור ${res.foods[idx].name}`,current);
  if(val===null)return;
  const n=Math.max(0,Math.min(res.foods[idx].max||1000,+val||0));
  res.amounts[idx]=n;res.totals=totalsFromAmounts(res.foods,res.amounts);state.result=res;renderMenu();
}
