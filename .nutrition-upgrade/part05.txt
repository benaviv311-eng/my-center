function goStep(n){
  if(n>1 && !state.targets){ calculateTargets(); if(!state.targets) return; }
  ['step1','step2','step3'].forEach((id,i)=>qs(id).classList.toggle('hidden',i!==n-1));
  document.querySelectorAll('.step').forEach((b,i)=>b.classList.toggle('active',i===n-1));
  window.scrollTo({top:0,behavior:'smooth'});
}

function calculateTargets(){
  const age=+qs('age').value, w=+qs('weight').value, h=+qs('height').value, sex=qs('sex').value;
  if(!age||!w||!h){alert('מלא גיל, משקל וגובה');return;}
  const bmr=10*w+6.25*h-5*age+(sex==='male'?5:-161);
  const factors=[0,1.2,1.35,1.55,1.75,1.9];
  let cal=bmr*factors[state.activity];
  if(state.goal==='loss') cal-=350;
  if(state.goal==='gain') cal+=250;
  if(state.goal==='performance') cal+=100;
  cal=Math.max(cal,1200);
  const baseProtein=[0,.9,1.1,1.4,1.6,1.8][state.activity];
  let pRate=baseProtein;
  if(state.goal==='loss') pRate=Math.max(pRate,1.7);
  if(state.goal==='gain') pRate=Math.max(pRate,1.7);
  if(state.goal==='performance') pRate=Math.max(pRate,1.5);
  const protein=Math.round(w*pRate);
  let fat=Math.round((cal*.28)/9);
  fat=Math.max(fat,Math.round(w*.6));
  let carbs=Math.round((cal-protein*4-fat*9)/4);
  carbs=Math.max(carbs,80);
  const fiber=Math.round(Math.max(25,cal/1000*14));
  const water=Math.round((w*(30+state.activity*1.5))/100)/10;
  state.targets={cal:Math.round(cal/10)*10,protein,carbs,fat,fiber,water};
  qs('tCal').textContent=state.targets.cal;
  qs('tProtein').textContent=protein;
  qs('tCarbs').textContent=carbs;
  qs('tFat').textContent=fat;
  qs('tFiber').textContent=fiber;
  qs('tWater').textContent=water;
  qs('targetsWrap').classList.remove('hidden');
  localStorage.setItem('nutritionProfile',JSON.stringify({
    age,w,h,sex,activity:state.activity,goal:state.goal,targets:state.targets
  }));
}

async function addFood(type){
  const input=qs(type+'Input'),text=input.value.trim();if(!text)return;
  if(state.foods[type].some(x=>normalize(x.raw)===normalize(text))){input.value='';return;}
  input.disabled=true;const oldPlaceholder=input.placeholder;input.placeholder='מחפש במאגר…';
  let food=findFood(text);if(!food){try{food=await searchGovFood(text)}catch(e){food=null}}
  if(food&&!DB.some(x=>x.id===food.id))DB.push(food);
  state.foods[type].push({raw:text,id:food?food.id:null});input.value='';input.disabled=false;input.placeholder=oldPlaceholder;renderChips(type);
}

function defaultsForMissing(list){
  const out=[...list];
  const haveCat=new Set(out.map(f=>f.cat));
  const defaults={protein:['chicken','egg','skyr'],carb:['rice','potato','oats'],fat:['tahini','oliveoil','almonds'],fiber:['lentils','broccoli','apple']};
  Object.entries(defaults).forEach(([cat,ids])=>{
    if(!haveCat.has(cat)) ids.forEach(id=>{const f=DB.find(x=>x.id===id);if(f&&!out.some(x=>x.id===id))out.push(f)});
  });
  return out;
}

function totalsFromAmounts(foods,amounts){
  const t={cal:0,protein:0,carbs:0,fat:0,fiber:0};
  foods.forEach((f,i)=>{
    const q=(amounts[i]||0)/100;
    t.cal+=f.k*q;t.protein+=f.p*q;t.carbs+=f.c*q;t.fat+=f.f*q;t.fiber+=f.fi*q;
  });
  return t;
}
function loss(t,target){
  const terms=[
    ['cal',1.7],['protein',1.7],['carbs',1],['fat',1],['fiber',1.5]
  ];
  return terms.reduce((sum,[k,w])=>{
    const goal=k==='protein'?target.protein:k==='carbs'?target.carbs:k==='fat'?target.fat:k==='fiber'?target.fiber:target.cal;
    const d=(t[k]-goal)/Math.max(goal,1);
    const over=(k==='cal'&&d>0)?1.7:1;
    return sum+w*over*d*d;
  },0);
}

function optimize(foods,target){
  let amounts=foods.map(()=>0);
  let current=totalsFromAmounts(foods,amounts);
  let currentLoss=loss(current,target);
  for(let iter=0;iter<1800;iter++){
    let best=null;
    for(let i=0;i<foods.length;i++){
      for(const step of [10,-10]){
        const next=Math.max(0,Math.min(foods[i].max||500,amounts[i]+step));
        if(next===amounts[i])continue;
        const trial=[...amounts];trial[i]=next;
        const tt=totalsFromAmounts(foods,trial);
        const L=loss(tt,target);
        if(!best||L<best.L)best={i,next,L,tt};
      }
    }
    if(!best||best.L>=currentLoss-1e-8)break;
    amounts[best.i]=best.next;current=best.tt;currentLoss=best.L;
  }
  amounts=amounts.map(a=>a<10?0:a);
  current=totalsFromAmounts(foods,amounts);
  return {foods,amounts,totals:current,score:loss(current,target)};
}

function issueList(res){
  const t=state.targets, a=res.totals;
  const issues=[];
  if(a.cal>t.cal*1.10)issues.push(`הקלוריות גבוהות בכ-${Math.round(a.cal-t.cal)} קק"ל`);
  if(a.cal<t.cal*.85)issues.push(`חסרות כ-${Math.round(t.cal-a.cal)} קלוריות`);
  if(a.protein<t.protein*.85)issues.push(`חסרים כ-${Math.round(t.protein-a.protein)} גרם חלבון`);
  if(a.fiber<t.fiber*.80)issues.push(`חסרים כ-${Math.round(t.fiber-a.fiber)} גרם סיבים`);
  if(a.fat>t.fat*1.25)issues.push(`השומן גבוה משמעותית מהיעד`);
  return issues;
}

function bestFiberSuggestions(excludedIds){
  return DB.filter(f=>f.fi>2&&!excludedIds.includes(f.id))
    .map(f=>({f,ratio:f.fi/Math.max(f.k,1)}))
    .sort((a,b)=>b.ratio-a.ratio).slice(0,3).map(x=>x.f);
}
